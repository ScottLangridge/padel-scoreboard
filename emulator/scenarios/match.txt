# A whole match, from power on: team A (remote a) beats team B 6-0 6-0.
# Along the way: the splash is skipped, a stranger's remote is ignored, a
# long press undoes a point, the ends change after set 1, and a golden point.

# Power on to the first point, step by step (`start` does all of this).
tap a                            # any press skips the splash...
wait log "bind: press the left remote"
expect log "splash skipped"      # ...and doesn't bind anything
wait 1
snap                             # PRESS / LEFT
tap a
wait log "press the right remote"
wait 0.7
tap b
wait log "select serve"
expect log "remote 0x005a5a51 is left (team A)"
expect log "remote 0x005a5a52 is right (team B)"
wait 1
tap a
wait idle
expect score "sets 0-0 games 0-0 points 0-0 serve A"
expect board                     # the fresh board: 0 0, games 0 0, A's serve triangle
  .........###........###.........
  ........#...#......#...#........
  ........#..##......#..##........
  ........#.#.#.####.#.#.#........
  ........##..#......##..#........
  ........#...#......#...#........
  .........###........###.........
  ................................
  ################################
  ..#..........................#..
  ..#...###..............###...#..
  ..#...#.#....#.........#.#...#..
  ..#...#.#....##........#.#...#..
  ..#...#.#....###.......#.#...#..
  ..#...###....####......###...#..
  ..#..........................#..

point x                          # a stranger's remote
expect log "unknown, ignored"
expect score "points 0-0"

# Set 1, with an undo in the first game.
point a x2
point b
expect score "points 2-1"
undo b                           # either remote's long press undoes the last point
expect log "undo"
expect score "points 2-0"
game a                           # finishes that game
expect score "games 1-0 points 0-0 serve B"
game a x5                        # the set: 6-0
expect log "swap ends"
snap                             # SWAP / ENDS

swap
expect score "sets 1-0 games 0-0 points 0-0 serve A [A on the right]"
expect pixel 31 12 on            # A's set is now tallied on the right
expect pixel 0 12 off

# Set 2: a golden point first.
point a x3
point b x3
expect score "points 3-3"
snap                             # GOLDEN
point b                          # the golden point decides the game
expect score "games 0-1 points 0-0"
set a
expect log "match: team A wins 6-0 6-1"
wait 6
snap                             # MATCH and the final score
